package main

import (
        "context"
        "errors"
        "log"
        "net/http"
        "os"
        "os/signal"
        "syscall"
        "time"

        "calculator/internal/httpapi"
)

func main() {
        if err := run(); err != nil {
                log.Fatal(err)
        }
}

func run() error {
        port := os.Getenv("PORT")
        if port == "" {
                port = "8080"
        }
        var static http.Handler
        if dir := os.Getenv("STATIC_DIR"); dir != "" {
                if _, err := os.Stat(dir + "/index.html"); err != nil {
                        return err
                }
                static = http.FileServer(http.Dir(dir))
        }
        server := &http.Server{
                Addr:              ":" + port,
                Handler:           httpapi.New(static),
                ReadHeaderTimeout: 5 * time.Second,
                ReadTimeout:       10 * time.Second,
                WriteTimeout:      10 * time.Second,
                IdleTimeout:       60 * time.Second,
        }
        ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
        defer stop()
        done := make(chan error, 1)
        go func() {
                log.Printf("Calculator listening on %s", server.Addr)
                done <- server.ListenAndServe()
        }()
        select {
        case err := <-done:
                if errors.Is(err, http.ErrServerClosed) {
                        return nil
                }
                return err
        case <-ctx.Done():
                shutdownCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
                defer cancel()
                return server.Shutdown(shutdownCtx)
        }
}

// Package calculator implements arithmetic independently of HTTP.
package calculator

import (
        "fmt"
        "math"
)

// Error is a validation or arithmetic error safe to expose to clients.
type Error struct {
        Code    string
        Message string
}

func (e *Error) Error() string { return e.Message }

// Calculate evaluates one operation using finite IEEE 754 numbers.
// Percentage means the first operand percent of the second operand.
func Calculate(operation string, operands []float64) (float64, error) {
        var arity int
        switch operation {
        case "add", "subtract", "multiply", "divide", "power", "percentage":
                arity = 2
        case "sqrt":
                arity = 1
        default:
                return 0, &Error{"INVALID_OPERATION", "Choose a supported operation."}
        }
        if len(operands) != arity {
                return 0, &Error{"INVALID_OPERANDS", fmt.Sprintf("This operation requires %d operand(s).", arity)}
        }
        for _, operand := range operands {
                if math.IsNaN(operand) || math.IsInf(operand, 0) {
                        return 0, &Error{"INVALID_OPERANDS", "Operands must be finite numbers."}
                }
        }

        a := operands[0]
        var result float64
        if operation == "sqrt" {
                if a < 0 {
                        return 0, &Error{"NEGATIVE_SQUARE_ROOT", "Square root requires a non-negative number."}
                }
                result = math.Sqrt(a)
        } else {
                b := operands[1]
                switch operation {
                case "add":
                        result = a + b
                case "subtract":
                        result = a - b
                case "multiply":
                        result = a * b
                case "divide":
                        if b == 0 {
                                return 0, &Error{"DIVISION_BY_ZERO", "Cannot divide by zero."}
                        }
                        result = a / b
                case "power":
                        if a == 0 && b < 0 {
                                return 0, &Error{"DIVISION_BY_ZERO", "Zero cannot be raised to a negative power."}
                        }
                        result = math.Pow(a, b)
                case "percentage":
                        // Divide the larger magnitude first to reduce intermediate overflow/underflow.
                        if math.Abs(a) >= math.Abs(b) {
                                result = (a / 100) * b
                        } else {
                                result = a * (b / 100)
                        }
                }
        }
        if math.IsNaN(result) {
                return 0, &Error{"INVALID_RESULT", "The operation has no real-valued result."}
        }
        if math.IsInf(result, 0) {
                return 0, &Error{"RESULT_OUT_OF_RANGE", "The result exceeds the supported numeric range."}
        }
        // Normalize negative zero for consistent JSON and display.
        if result == 0 {
                return 0, nil
        }
        return result, nil
}

// Package httpapi translates HTTP requests into calculator calls.
package httpapi

import (
        "encoding/json"
        "errors"
        "io"
        "mime"
        "net/http"

        "calculator/internal/calculator"
)

const maxBodyBytes = 4096

type calculationRequest struct {
        Operation string     `json:"operation"`
        Operands  []*float64 `json:"operands"`
}

type calculationResponse struct {
        Result float64 `json:"result"`
}

type errorDetail struct {
        Code    string `json:"code"`
        Message string `json:"message"`
}

type errorResponse struct {
        Error errorDetail `json:"error"`
}

// New builds a handler with an optional static frontend for production.
func New(static http.Handler) http.Handler {
        mux := http.NewServeMux()
        mux.HandleFunc("/api/v1/calculate", calculate)
        mux.HandleFunc("/api/v1/health", health)
        mux.HandleFunc("/api/", func(w http.ResponseWriter, r *http.Request) {
                writeError(w, http.StatusNotFound, "NOT_FOUND", "API endpoint not found.")
        })
        if static != nil {
                mux.Handle("/", static)
        } else {
                mux.HandleFunc("/", func(w http.ResponseWriter, r *http.Request) {
                        writeError(w, http.StatusNotFound, "NOT_FOUND", "Endpoint not found.")
                })
        }
        return mux
}

func health(w http.ResponseWriter, r *http.Request) {
        if r.Method != http.MethodGet {
                w.Header().Set("Allow", http.MethodGet)
                writeError(w, http.StatusMethodNotAllowed, "METHOD_NOT_ALLOWED", "Use GET for this endpoint.")
                return
        }
        writeJSON(w, http.StatusOK, map[string]string{"status": "ok"})
}

func calculate(w http.ResponseWriter, r *http.Request) {
        if r.Method != http.MethodPost {
                w.Header().Set("Allow", http.MethodPost)
                writeError(w, http.StatusMethodNotAllowed, "METHOD_NOT_ALLOWED", "Use POST for this endpoint.")
                return
        }
        mediaType, _, err := mime.ParseMediaType(r.Header.Get("Content-Type"))
        if err != nil || mediaType != "application/json" {
                writeError(w, http.StatusUnsupportedMediaType, "UNSUPPORTED_MEDIA_TYPE", "Send application/json.")
                return
        }
        r.Body = http.MaxBytesReader(w, r.Body, maxBodyBytes)
        defer r.Body.Close()
        decoder := json.NewDecoder(r.Body)
        decoder.DisallowUnknownFields()
        var request *calculationRequest
        if err := decoder.Decode(&request); err != nil {
                writeDecodeError(w, err)
                return
        }
        if request == nil {
                writeError(w, http.StatusBadRequest, "INVALID_JSON", "Send a JSON object.")
                return
        }
        // A second decode must reach EOF; concatenated JSON values are rejected.
        if err := decoder.Decode(new(any)); err != io.EOF {
                writeDecodeError(w, err)
                return
        }
        operands := make([]float64, 0, len(request.Operands))
        for _, operand := range request.Operands {
                if operand == nil {
                        writeError(w, http.StatusBadRequest, "INVALID_OPERANDS", "Operands must be numbers, not null.")
                        return
                }
                operands = append(operands, *operand)
        }
        result, err := calculator.Calculate(request.Operation, operands)
        if err != nil {
                var calcErr *calculator.Error
                if errors.As(err, &calcErr) {
                        writeError(w, http.StatusBadRequest, calcErr.Code, calcErr.Message)
                        return
                }
                writeError(w, http.StatusInternalServerError, "INTERNAL_ERROR", "Calculation failed unexpectedly.")
                return
        }
        writeJSON(w, http.StatusOK, calculationResponse{Result: result})
}

func writeDecodeError(w http.ResponseWriter, err error) {
        var tooLarge *http.MaxBytesError
        if errors.As(err, &tooLarge) {
                writeError(w, http.StatusRequestEntityTooLarge, "BODY_TOO_LARGE", "Request body exceeds 4096 bytes.")
                return
        }
        writeError(w, http.StatusBadRequest, "INVALID_JSON", "Send one JSON object with an operation and numeric operands.")
}

func writeError(w http.ResponseWriter, status int, code, message string) {
        writeJSON(w, status, errorResponse{Error: errorDetail{Code: code, Message: message}})
}

func writeJSON(w http.ResponseWriter, status int, value any) {
        w.Header().Set("Content-Type", "application/json; charset=utf-8")
        w.Header().Set("X-Content-Type-Options", "nosniff")
        w.Header().Set("Cache-Control", "no-store")
        w.WriteHeader(status)
        // All response values are JSON-serializable and finite.
        _ = json.NewEncoder(w).Encode(value)
}
